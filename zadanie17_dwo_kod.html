<script type="text/javascript">
 function pole_dwo()
 {
var a_dwo= parseFloat(document.getElementById('dlugosc').value);
var b_dwo= a_dwo*a_dwo;
 document.getElementById("nazwa").innerHTML ="Pole kwadratu= "+b_dwo.toFixed(2)+" cm^2";
 }
 function obwod_dwo()
 {
var a_dwo= parseFloat(document.getElementById('dlugosc').value);
var b_dwo= 4*a_dwo;
 document.getElementById("nazwa").innerHTML ="Obwód kwadratu= "+b_dwo.toFixed(2)+" cm";
 }
</script>
