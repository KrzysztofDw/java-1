{
 r_dwo=parseFloat(r)

 Pc_wynik=2*Math.PI*r_dwo*r_dwo+2*Math.PI*r_dwo*h_dwo
 document.getElementsByName('pole')[0].value=Pc_wynik.toFixed(2);
